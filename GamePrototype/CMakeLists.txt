cmake_minimum_required(VERSION 3.20)
project(AdaptiveSocialSurvivalTPS LANGUAGES CXX)
include(CTest)
add_library(survival_core STATIC cpp/core/src/game_session.cpp cpp/core/src/simulation.cpp)
add_library(Survival::Core ALIAS survival_core)
target_include_directories(survival_core PUBLIC cpp/core/include)
target_compile_features(survival_core PUBLIC cxx_std_20)
set_target_properties(survival_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
add_library(survival_synthetic STATIC cpp/core/src/synthetic.cpp)
target_link_libraries(survival_synthetic PUBLIC Survival::Core)
add_executable(GameRuntime cpp/runtime/main.cpp cpp/runtime/transport.cpp)
target_include_directories(GameRuntime PRIVATE third_party)
target_link_libraries(GameRuntime PRIVATE survival_synthetic)
if(BUILD_TESTING)
  add_executable(core_tests cpp/tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE Survival::Core)
  add_test(NAME backend_independent_core COMMAND core_tests)
  add_executable(scenario_tests cpp/tests/scenario_tests.cpp)
  target_link_libraries(scenario_tests PRIVATE survival_synthetic)
  add_test(NAME seeded_scenarios COMMAND scenario_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME python_and_protocol COMMAND ${Python3_EXECUTABLE} -m unittest discover -s python/tests -v)
    set_tests_properties(python_and_protocol PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} ENVIRONMENT "SURVIVAL_RUNTIME=$<TARGET_FILE:GameRuntime>")
  endif()
endif()
